\section{Selected Publications}
{\small
\enlargethispage{2.2\baselineskip}

\Needspace{5\baselineskip}
\cvitem{\textbf{Subhransu S.\ Bhattacharjee}, Dylan Campbell \& Rahul Shome}{\textit{Believing is Seeing: Unobserved Object Detection using Generative Models}. \textit{IEEE/CVF Conference on Computer Vision and Pattern Recognition (CVPR)}, 2025, Nashville, Tennessee, USA. \href{https://1ssb.github.io/UOD/}{Project Page}.
\\ \textit{Defines 2D, 2.5D, and 3D unobserved object detection from a single RGB view with diffusion/VLM pipelines and tailored metrics; the 3D-diffusion method outperforms all applicable methods, including an in-frustum oracle, on the 3D task.}
}

\Needspace{4\baselineskip}
\cvitem{\textbf{Subhransu S.\ Bhattacharjee}, Dylan Campbell \& Rahul Shome}{\textit{FlatLands: Generative Floormap Completion From a Single Egocentric View}. \textit{European Conference on Computer Vision (ECCV)}, 2026, Malmö, Sweden. Springer. \href{https://1ssb.github.io/Flat_Lands/}{Project Page}.
\\ \textit{Releases FlatLands: 270,575 observations from 17,656 real metric indoor scenes across six datasets.}
\\ \textit{Compares 11 methods, showing stochastic generators improve completion fidelity and capture layout uncertainty.}
}

\Needspace{6\baselineskip}
\cvitem{\textbf{Subhransu S.\ Bhattacharjee}, Hao Lu, Dylan Campbell \& Rahul Shome}{\textit{MatterDoor: Sampling Zero-shot Spatio\mbox{-}semantic Priors using Generative Models}. \textit{arXiv:2510.11014}, 2025. Under review; presented at the 2026 Robotics: Science and Systems FM4RoboPlan workshop in Sydney. \href{https://arxiv.org/pdf/2510.11014}{Preprint}.
\\
\textit{Samples environment priors from partial observations using pretrained generative models.}
\\
\textit{Produces RGB-D point-cloud hypotheses for occupancy, target semantics, and configuration-space planning.}
}

\Needspace{5\baselineskip}
\cvitem{\textbf{Subhransu S.\ Bhattacharjee} \& Ian Petersen}{\textit{Analysis of the Whiplash Gradient Descent Dynamics}. \textit{Asian Journal of Control} (Special Issue), Wiley, 2023. DOI: \href{https://onlinelibrary.wiley.com/doi/10.1002/asjc.3153}{10.1002/asjc.3153}.
\\
\textit{Introduces a closed-loop inertial optimiser with adaptive damping, deterministic saddle-point escape, and symplectic Lyapunov convergence analysis.}
\textit{Establishes polynomial and exponential convergence rates for adaptive inertial gradient dynamics. \href{https://github.com/1ssb/Whiplash}{Code}.}
\\
\textit{Conference papers: ASCC 2022, Jeju Island (\href{https://doi.org/10.23919/ASCC56756.2022.9828104}{DOI}); ANZCC 2021, Gold Coast (\href{https://doi.org/10.1109/ANZCC53563.2021.9628258}{DOI}).}
}
